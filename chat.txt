```java
package com.example;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.sql.SQLException;
import java.util.List;

/**
 * Main application to scan a folder, store file details, and detect duplicates.
 * Writes to SQLite database in dups/ folder.
 */
public class App {

    private static final String DB_PATH = "dups/duplicates.db";

    public static void main(String[] args) {
        // Ensure dups folder exists
        Path dbFolder = Paths.get("dups");
        try {
            Files.createDirectories(dbFolder);
        } catch (IOException e) {
            System.err.println("Failed to create dups folder: " + e.getMessage());
            return;
        }

        String pathToScan = args.length > 0 ? args[0] : ".";
        System.out.println("Scanning directory: " + pathToScan);

        try {
            // 1. Scan and store files
            FileScanner scanner = new FileScanner(DB_PATH);
            int filesAdded = scanner.scanFolder(pathToScan);
            System.out.println("Scanned and stored details of " + filesAdded + " files.");

            // 2. Find duplicates
            DuplicateFinder finder = new DuplicateFinder(DB_PATH);
            List<DuplicateFinder.FileGroup> duplicates = finder.findDuplicates();

            System.out.println("\nFound " + duplicates.size() + " duplicate groups:");
            if (duplicates.isEmpty()) {
                System.out.println("No duplicates found.");
            } else {
                // Print sorted by largest first
                duplicates.stream()
                    .sorted((a, b) -> Long.compare(b.getTotalSize(), a.getTotalSize()))
                    .forEach(group -> {
                        System.out.println("\nGroup - Size: " + group.getTotalSize() + " bytes, Count: " + group.getFiles().size());
                        group.getFiles().forEach(f ->
                            System.out.println("  - " + f.getFullPath() + " (" + f.getSize() + " bytes)")
                        );
                    });
            }

        } catch (SQLException e) {
            System.err.println("Database error: " + e.getMessage());
            e.printStackTrace();
        } catch (IOException e) {
            System.err.println("File system error: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
```

```java
package com.example;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;

/**
 * Handles database creation, schema and utility methods.
 */
public class DatabaseManager {

    private static final String JDBC_URL_PREFIX = "jdbc:sqlite:";
    private final String dbPath;

    public DatabaseManager(String dbPath) {
        this.dbPath = dbPath;
    }

    public Connection getConnection() throws SQLException {
        return DriverManager.getConnection(JDBC_URL_PREFIX + dbPath);
    }

    /**
     * Creates tables if they don't exist.
     */
    public void initializeSchema() throws SQLException {
        try (Connection conn = getConnection();
             Statement stmt = conn.createStatement()) {

            String fileTable = """
                CREATE TABLE IF NOT EXISTS files (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    path TEXT UNIQUE,
                    size INTEGER NOT NULL,
                    last_modified LONG NOT NULL,
                    md5_hash TEXT
                )""";

            String dupTable = """
                CREATE TABLE IF NOT EXISTS duplicates (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    hash TEXT NOT NULL,
                    file_id INTEGER NOT NULL,
                    FOREIGN KEY (file_id) REFERENCES files(id)
                )""";

            stmt.execute(fileTable);
            stmt.execute(dupTable);
        }
    }

    /**
     * Returns true if a file path already exists in the database.
     */
    public boolean fileExists(String path) throws SQLException {
        String sql = "SELECT 1 FROM files WHERE path = ?";
        try (Connection conn = getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setString(1, path);
            try (ResultSet rs = stmt.executeQuery()) {
                return rs.next();
            }
        }
    }

    /**
     * Adds or updates a file record (idempotent).
     */
    public void upsertFile(String path, long size, long lastModified) throws SQLException {
        String checkSql = "SELECT id FROM files WHERE path = ?";
        String insertSql = "INSERT INTO files (path, size, last_modified, md5_hash) VALUES (?, ?, ?, NULL)";
        String updateSql = "UPDATE files SET size = ?, last_modified = ?, md5_hash = NULL WHERE path = ?";

        try (Connection conn = getConnection();
             PreparedStatement checkStmt = conn.prepareStatement(checkSql);
             PreparedStatement insertStmt = conn.prepareStatement(insertSql);
             PreparedStatement updateStmt = conn.prepareStatement(updateSql)) {

            checkStmt.setString(1, path);
            try (ResultSet rs = checkStmt.executeQuery()) {
                if (rs.next()) {
                    // File exists, update it
                    updateStmt.setLong(1, size);
                    updateStmt.setLong(2, lastModified);
                    updateStmt.setString(3, path);
                    updateStmt.executeUpdate();
                } else {
                    // File does not exist, insert it
                    insertStmt.setString(1, path);
                    insertStmt.setLong(2, size);
                    insertStmt.setLong(3, lastModified);
                    insertStmt.executeUpdate();
                }
            }
        }
    }

    /**
     * Clears all data in tables for fresh scan.
     */
    public void clearData() throws SQLException {
        String deleteFilesSql = "DELETE FROM files";
        String deleteDuplicatesSql = "DELETE FROM duplicates";
        try (Connection conn = getConnection();
             Statement stmt = conn.createStatement()) {
            stmt.execute(deleteFilesSql);
            stmt.execute(deleteDuplicatesSql);
        }
    }
}
```

```java
package com.example;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.sql.SQLException;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * Scans a directory tree and stores file details in the database.
 */
public class FileScanner {

    private final DatabaseManager dbManager;

    public FileScanner(String dbPath) {
        this.dbManager = new DatabaseManager(dbPath);
        try {
            dbManager.initializeSchema();
        } catch (SQLException e) {
            System.err.println("Failed to initialize database: " + e.getMessage());
            throw new RuntimeException(e);
        }
    }

    public int scanFolder(String folderPath) throws IOException, SQLException {
        Path folder = Paths.get(folderPath);
        if (!Files.exists(folder) || !Files.isDirectory(folder)) {
            throw new IllegalArgumentException("Folder does not exist or is not a directory: " + folderPath);
        }

        // For reruns: clear existing data if you want fresh scan
        // Or skip existing — here we choose to update existing
        // dbManager.clearData(); // Uncomment to start fresh

        AtomicInteger count = new AtomicInteger();
        Files.walk(folder)
            .filter(Files::isRegularFile)
            .forEach(path -> {
                try {
                    if (!dbManager.fileExists(path.toString())) {
                        dbManager.upsertFile(
                            path.toString(),
                            Files.size(path),
                            Files.getLastModifiedTime(path).toMillis()
                        );
                    } else {
                        // Update if changed
                        dbManager.upsertFile(
                            path.toString(),
                            Files.size(path),
                            Files.getLastModifiedTime(path).toMillis()
                        );
                    }
                    count.incrementAndGet();
                } catch (SQLException e) {
                    System.err.println("Failed to process " + path + ": " + e.getMessage());
                }
            });

        return count.get();
    }
}
```

```java
package com.example;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.*;
import java.util.stream.Collectors;

/**
 * Finds duplicate files by size (fast) or hash (robust, but more work).
 * This implementation uses size grouping, then MD5 for verification.
 * For simplicity, this version uses size grouping only (fast and works for many cases).
 */
public class DuplicateFinder {

    public static class FileInfo {
        private final String fullPath;
        private final long size;

        public FileInfo(String fullPath, long size) {
            this.fullPath = fullPath;
            this.size = size;
        }

        public String getFullPath() { return fullPath; }
        public long getSize() { return size; }
    }

    public static class FileGroup {
        private final long totalSize;
        private final List<FileInfo> files;

        public FileGroup(long totalSize, List<FileInfo> files) {
            this.totalSize = totalSize;
            this.files = files;
        }

        public long getTotalSize() { return totalSize; }
        public List<FileInfo> getFiles() { return files; }
    }

    private final DatabaseManager dbManager;

    public DuplicateFinder(String dbPath) {
        this.dbManager = new DatabaseManager(dbPath);
    }

    /**
     * Finds duplicate files by size (fast), and optionally hash (slower, more accurate).
     * This version uses size grouping only for simplicity.
     */
    public List<FileGroup> findDuplicates() throws SQLException {
        List<FileGroup> duplicates = new ArrayList<>();

        String sql = """
            SELECT size, GROUP_CONCAT(id || ':' || path, '|') as files
            FROM files
            WHERE size > 0
            GROUP BY size
            HAVING COUNT(*) > 1""";

        try (Connection conn = dbManager.getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql);
             ResultSet rs = stmt.executeQuery()) {

            while (rs.next()) {
                long size = rs.getLong("size");
                String filesStr = rs.getString("files");

                List<FileInfo> files = Arrays.stream(filesStr.split("\\|"))
                    .map(part -> {
                        String[] parts = part.split(":", 2);
                        return new FileInfo(parts.length == 2 ? parts[1] : parts[0], size);
                    })
                    .collect(Collectors.toList());

                if (files.size() > 1) {
                    duplicates.add(new FileGroup(files.stream().mapToLong(FileInfo::getSize).sum(), files));
                }
            }
        }

        return duplicates;
    }

    // Optional: Add hash-based verification for accuracy
    public void computeAndStoreHashes() throws SQLException, NoSuchAlgorithmException, IOException {
        String selectSql = "SELECT id, path FROM files WHERE md5_hash IS NULL";
        String updateSql = "UPDATE files SET md5_hash = ? WHERE id = ?";

        try (Connection conn = dbManager.getConnection();
             PreparedStatement selectStmt = conn.prepareStatement(selectSql);
             PreparedStatement updateStmt = conn.prepareStatement(updateSql)) {

            selectStmt.setMaxRows(1000); // Limit for demo
            try (ResultSet rs = selectStmt.executeQuery()) {
                java.security.MessageDigest md = java.security.MessageDigest.getInstance("MD5");
                while (rs.next()) {
                    int id = rs.getInt("id");
                    String path = rs.getString("path");

                    try (java.io.InputStream is = new java.io.FileInputStream(path)) {
                        byte[] buffer = new byte[8192];
                        int len;
                        while ((len = is.read(buffer)) != -1) {
                            md.update(buffer, 0, len);
                        }
                        byte[] digest = md.digest();
                        StringBuilder sb = new StringBuilder();
                        for (byte b : digest) {
                            sb.append(String.format("%02x", b));
                        }
                        updateStmt.setString(1, sb.toString());
                        updateStmt.setInt(2, id);
                        updateStmt.addBatch();
                    }
                }
                updateStmt.executeBatch();
            }
        }
    }
}
```

```java
            dbManager.clearData();

        assertDoesNotThrow(() -> dbManager.initializeSchema());
        // Verify tables exist
        try (var conn = dbManager.getConnection();
             var stmt = conn.createStatement();
             var rs = stmt.executeQuery("SELECT name FROM sqlite_master WHERE type='table'")) {

            boolean hasFiles = false;
            boolean hasDuplicates = false;
            while (rs.next()) {
                String table = rs.getString("name");
                if ("files".equals(table)) hasFiles = true;
                if ("duplicates".equals(table)) hasDuplicates = true;
            }

            assertTrue(hasFiles, "files table should exist");
            assertTrue(hasDuplicates, "duplicates table should exist");
        } catch (SQLException e) {
            fail("Failed to verify tables", e);
        }
    }

    @Test
    void testUpsertFile() throws SQLException {
        dbManager.upsertFile("/tmp/test.txt", 1024, System.currentTimeMillis());
        assertTrue(dbManager.fileExists("/tmp/test.txt"));

        dbManager.upsertFile("/tmp/test.txt", 2048, System.currentTimeMillis());
        // Should still exist, but size should be updated
        assertTrue(dbManager.fileExists("/tmp/test.txt"));
    }

    @Test
    void testClearData() throws SQLException {
        dbManager.upsertFile("/tmp/test.txt", 1024, System.currentTimeMillis());
        dbManager.clearData();
        assertFalse(dbManager.fileExists("/tmp/test.txt"));
    }

    @Test
    void testFileExists() throws SQLException {
        dbManager.upsertFile("/tmp/unique.txt", 500, System.currentTimeMillis());
        assertTrue(dbManager.fileExists("/tmp/unique.txt"));
        assertFalse(dbManager.fileExists("/tmp/nonexistent.txt"));
    }

    // Cleanup after tests
    void tearDown() throws IOException {
        Path dbFile = Path.of(TEST_DB);
        if (Files.exists(dbFile)) {
            Files.delete(dbFile);
        }
    }
}
```

```java
package com.example;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import java.io.IOException;
import java.nio.file.*;
import java.sql.SQLException;

import static org.junit.jupiter.api.Assertions.*;

class FileScannerTest {

    private static final String TEST_DB = "dups/test_scan.db";
    private Path testDir;
    private FileScanner scanner;

    @BeforeEach
    void setUp() throws IOException {
        testDir = Files.createTempDirectory("file-scanner-test");
        Path dbPath = Path.of("dups");
        Files.createDirectories(dbPath);
        scanner = new FileScanner(TEST_DB);
    }

    @Test
    void testScanExistingDirectory() throws IOException, SQLException {
        // Create test files
        Path file1 = Files.createFile(testDir.resolve("file1.txt"));
        Files.write(file1, "Hello".getBytes());

        Path file2 = Files.createFile(testDir.resolve("file2.txt"));
        Files.write(file2, "World!".getBytes());

        int count = scanner.scanFolder(testDir.toString());
        assertEquals(2, count, "Should find 2 files");
    }

    @Test
    void testScanNonExistentDirectory() {
        assertThrows(IllegalArgumentException.class, () -> {
            scanner.scanFolder("/nonexistent/dir");
        });
    }

    @Test
    void testRerunScan() throws IOException, SQLException {
        Path file = Files.createFile(testDir.resolve("test.txt"));
        Files.write(file, "First".getBytes());

        // First scan
        scanner.scanFolder(testDir.toString());

        // Change file
        Files.write(file, "Second content".getBytes());

        // Rerun scan (update existing)
        int count = scanner.scanFolder(testDir.toString());
        assertEquals(1, count, "Should find 1 file (updated)");
    }

    @Test
    void testClearDatabase() throws SQLException {
        scanner.scanFolder(testDir.toString());
        DatabaseManager db = new DatabaseManager(TEST_DB);
        db.clearData();
        // Database should be empty
        try (var conn = db.getConnection();
             var rs = conn.createStatement().executeQuery("SELECT COUNT(*) FROM files")) {
            rs.next();
            assertEquals(0, rs.getInt(1));
        }
    }

    // Cleanup
    void tearDown() throws IOException {
        deleteRecursively(testDir);
        Path dbFile = Path.of(TEST_DB);
        if (Files.exists(dbFile)) {
            Files.delete(dbFile);
        }
    }

    private void deleteRecursively(Path path) throws IOException {
        if (Files.isDirectory(path)) {
            try (var stream = Files.list(path)) {
                stream.forEach(p -> {
                    try {
                        deleteRecursively(p);
                    } catch (IOException e) {
                        // ignore
                    }
                });
            }
        }
        Files.deleteIfExists(path);
    }
}
```
